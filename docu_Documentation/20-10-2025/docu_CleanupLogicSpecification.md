## 📋 Prehľad

Implementácia time-based cleanup mechanizmu s grace period pre prázdne riadky. Cleanup rozlišuje medzi:
- **Auto-generated prázdne riadky** → mazanie okamžité
- **User-inserted prázdne riadky** → mazanie po grace period (3 minúty default)

### KRITICKÁ ZMENA POŽIADAVKY

❌ **PÔVODNÁ POŽIADAVKA (NESPRÁVNA):**
- Odstrániť cleanup volania z Import/Add/Paste/SmartDelete

✅ **FINÁLNA POŽIADAVKA (SPRÁVNA):**
- **ZACHOVAŤ** cleanup volania po Import/Add/Paste/SmartDelete
- **PRIDAŤ** cleanup volanie pred Export

internal sealed record RowManagementConfiguration
{
    internal bool EnableAutoExpand { get; init; } = true;
    internal bool EnableSmartDelete { get; init; } = true;
    internal bool AlwaysKeepLastEmpty { get; init; } = true;

    /// <summary>
    /// Čas pred ktorým sa user-inserted prázdne riadky môžu zmazať.
    /// Default: 3 minúty (MENEJ ako 5 minút!)
    /// Set TimeSpan.MaxValue = nikdy nemaž user rows
    /// </summary>
    internal TimeSpan UserEmptyRowGracePeriod { get; init; } = TimeSpan.FromMinutes(3);

    /// <summary>
    /// Vždy mazať auto-generated prázdne riadky (okrem posledného).
    /// Default: true
    /// </summary>
    internal bool CleanupAutoGeneratedEmptyRows { get; init; } = true;
}

- `"__rowId"` → string (existuje)
- `"__lastModified"` → DateTime (NOVÉ)
- `"__creationType"` → string {"UserInserted", "AutoGenerated", "Imported"} (NOVÉ)

#### C) Updatovať EnsureMinRowsAndLastEmptyAsync

**Súbor:** SmartOperationService.cs (line 777-839)

**NOVÁ LOGIKA:**

```csharp
internal async Task EnsureMinRowsAndLastEmptyAsync(
    RowManagementConfiguration config,
    IReadOnlyDictionary<string, object?>? templateRow,
    CancellationToken cancellationToken)
{
    // STEP 1: Stream všetky riadky a nájdi prázdne na zmazanie
    var emptyRowsToDelete = new List<string>();

    await foreach (var batch in _rowStore.StreamRowsAsync(false, false, 1000, cancellationToken))
    {
        foreach (var row in batch)
        {
            if (!IsRowEmpty(row))
                continue; // Nie prázdny, skip

            // Získaj metadata
            var creationType = GetCreationType(row);
            var lastModified = GetLastModified(row);
            var age = DateTime.UtcNow - lastModified;

            // Rozhodovacia logika
            bool shouldDelete = false;

            // 1. Auto-generated → zmaž ak povolené
            if (creationType == "AutoGenerated" && config.CleanupAutoGeneratedEmptyRows)
            {
                shouldDelete = true;
            }
            // 2. User-inserted alebo Imported → zmaž len po grace period
            else if ((creationType == "UserInserted" || creationType == "Imported")
                     && age > config.UserEmptyRowGracePeriod)
            {
                shouldDelete = true;
            }

            if (shouldDelete)
            {
                var rowId = row["__rowId"]?.ToString();
                if (rowId != null)
                    emptyRowsToDelete.Add(rowId);
            }
        }
    }

    // Zmaž prázdne riadky
    if (emptyRowsToDelete.Any())
    {
        await _rowStore.RemoveRowsAsync(emptyRowsToDelete, cancellationToken);
        _logger.LogInformation("Deleted {Count} empty rows (grace period applied)",
            emptyRowsToDelete.Count);
    }

    // STEP 2: Ensure last empty row (AlwaysKeepLastEmpty)
    if (config.AlwaysKeepLastEmpty)
    {
        var rowCount = await _rowStore.GetRowCountAsync(cancellationToken);

        // Check if last row is empty
        bool needsEmptyRow = true;
        if (rowCount > 0)
        {
            // Get last row (streaming)
            await foreach (var batch in _rowStore.StreamRowsAsync(false, false, 1000, cancellationToken))
            {
                var lastRow = batch.LastOrDefault();
                if (lastRow != null && IsRowEmpty(lastRow))
                {
                    needsEmptyRow = false;
                    break;
                }
            }
        }
        else
        {
            needsEmptyRow = true; // Grid is empty, need 1 row
        }

        // Create empty row
        if (needsEmptyRow)
        {
            var newEmptyRow = CreateEmptyRow(templateRow);

            // Set metadata: AUTO-GENERATED
            ((Dictionary<string, object?>)newEmptyRow)["__creationType"] = "AutoGenerated";
            ((Dictionary<string, object?>)newEmptyRow)["__lastModified"] = DateTime.UtcNow;

            await _rowStore.AppendRowsAsync(new[] { newEmptyRow }, cancellationToken);
            _logger.LogInformation("Added last empty row (auto-generated)");
        }
    }
}

// Helper metódy
private string GetCreationType(IReadOnlyDictionary<string, object?> row)
{
    if (row.TryGetValue("__creationType", out var value))
        return value?.ToString() ?? "AutoGenerated";
    return "AutoGenerated"; // Default pre staré riadky
}

private DateTime GetLastModified(IReadOnlyDictionary<string, object?> row)
{
    if (row.TryGetValue("__lastModified", out var value) && value is DateTime dt)
        return dt;
    return DateTime.MinValue; // Staré riadky → okamžite zmazateľné
}

private bool IsRowEmpty(IReadOnlyDictionary<string, object?> row)
{
    return row
        .Where(kvp => !kvp.Key.StartsWith("__")) // Skip metadata
        .All(kvp => kvp.Value == null || string.IsNullOrWhiteSpace(kvp.Value.ToString()));
}

private IReadOnlyDictionary<string, object?> CreateEmptyRow(IReadOnlyDictionary<string, object?>? templateRow)
{
    var emptyRow = new Dictionary<string, object?>();

    if (templateRow != null)
    {
        foreach (var key in templateRow.Keys)
        {
            // Skip metadata (IRowStore assigns new __rowId)
            if (key.StartsWith("__"))
                continue;

            emptyRow[key] = null; // Empty value
        }
    }

    return emptyRow;
}
```
